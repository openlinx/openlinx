# Every runtime dependency of OpenLinx, direct or transitive, with the reason it is here.
# tests/test_dependencies.py fails when one is missing. Add a new one in the commit that adds it.
typer  # the CLI framework
annotated-doc  # typer: documentation for Annotated parameters
colorama  # typer, on Windows only: console colours
rich  # table output for people; typer's help
markdown-it-py  # rich: Markdown rendering
mdurl  # markdown-it-py: URL parsing
pygments  # rich: syntax highlighting
shellingham  # typer: detects the shell
aiohttp  # async HTTP transport for our own crawl engine (ADR-0001)
aiohappyeyeballs  # aiohttp: Happy Eyeballs connections
aiosignal  # aiohttp: callback lists
attrs  # aiohttp: its own classes
frozenlist  # aiohttp and aiosignal: immutable lists
multidict  # aiohttp and yarl: headers in order, with repeated names
propcache  # aiohttp and yarl: cached properties
yarl  # aiohttp: URLs, built from the normalised URL as given (design D13 of 020-add-crawl-core)
idna  # host names as aiohttp encodes them, UTS46 (design D13 of 020-add-crawl-core)
msgspec  # frozen DiscoveredLink and page-preparation records cross the planned 026-add-parse-pool process boundary (ADR-0006, 020-add-crawl-core D5)
orjson  # JSON output
publicsuffixlist  # registrable domains for the www rule and --scope subdomains, bundled, never downloaded (design D4 of 020-add-crawl-core)
pydantic-settings  # openlinx.toml settings
pydantic  # pydantic-settings: the settings models
pydantic-core  # pydantic: validation
annotated-types  # pydantic: constraint metadata
typing-extensions  # pydantic, pydantic-core, typing-inspection, sqlalchemy, aiohttp and aiosignal: typing backports
typing-inspection  # pydantic and pydantic-settings: runtime type inspection
python-dotenv  # pydantic-settings: the .env source, which OpenLinx drops (010-add-workspace-and-cli-contract D2)
selectolax  # the HTML parser of the link extractor, lexbor backend
protego  # the robots.txt parser
sqlalchemy  # the crawl store: SQLite through SQLAlchemy 2 Core (ADR-0003)
brotlicffi  # the br decoder of read_body, bounded in output (design D1 of 022-add-fetch-policy)
cffi  # BrotliCFFI: the interface to the native decoder
pycparser  # cffi: C declarations
backports.zstd  # read_body's bounded zstd decoder on Python 3.12 and 3.13
truststore  # TLS verification against the OS trust store, imported only from openlinx.crawl.fetch and scripts/tls_handshake_speed.py (design D5 of 022-add-fetch-policy)
certifi  # the CA bundle when truststore cannot be loaded, imported only from openlinx.crawl.fetch and scripts/tls_handshake_speed.py (design D5 of 022-add-fetch-policy)
